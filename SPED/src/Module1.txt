Sub NewRow_Clean()
    Dim ws As Worksheet
    Dim r As Long
    
    Set ws = ActiveSheet
    r = ActiveCell.Row
    If r < 2 Then
        MsgBox "Please select a cell below row 1.", vbExclamation
        Exit Sub
    End If
    
    ' Confirm new row insert
    If MsgBox("Do you want to add a new row?", vbYesNo) = vbNo Then Exit Sub
    
    ' Insert a row at current selection
    ws.Rows(r).Insert Shift:=xlDown, CopyOrigin:=xlFormatFromLeftOrAbove
    
    ' Confirm account link
    If MsgBox("Do you want to link an account as well?", vbYesNo) = vbNo Then Exit Sub
    
    ' Copy row above and paste into the newly inserted row
    ws.Rows(r - 1).Copy ws.Rows(r)
    
    Application.CutCopyMode = False
End Sub

Sub DeleteRow()
    ' Confirm before deleting
    If MsgBox("Do you want to delete the current row?", vbYesNo + vbQuestion, "Delete Row") = vbNo Then Exit Sub
    
    ' Make sure a cell is selected
    If Not TypeName(Selection) = "Range" Then
        MsgBox "Please select a cell inside the row you want to delete.", vbExclamation
        Exit Sub
    End If
    
    ' Delete the row of the active cell
    ActiveCell.EntireRow.Delete
End Sub

Sub CopyCurrentSheetWithName()

    Dim ws As Worksheet
    Dim newName As String
    Dim nameExists As Boolean
    Dim sh As Worksheet
    
    Set ws = ActiveSheet
    
    ' Ask user for sheet name
    newName = InputBox("Enter the name for the new sheet:", "New Sheet Name")
    
    ' If user cancels or leaves blank
    If Trim(newName) = "" Then
        MsgBox "Operation cancelled.", vbExclamation
        Exit Sub
    End If
    
    ' Check if name already exists
    nameExists = False
    For Each sh In ThisWorkbook.Worksheets
        If LCase(sh.Name) = LCase(newName) Then
            nameExists = True
            Exit For
        End If
    Next sh
    
    If nameExists Then
        MsgBox "A sheet with this name already exists.", vbCritical
        Exit Sub
    End If
    
    ' Copy sheet after current sheet
    ws.Copy After:=ws
    
    ' Rename the copied sheet
    ActiveSheet.Name = newName
    
    MsgBox "Sheet created successfully!", vbInformation

End Sub

Public Sub UiDock(ByVal frm As Object, ByVal side As String)
    ' Place a form on the monitor Excel is showing on (screen coordinates in points).
    Dim l As Single
    On Error Resume Next
    frm.StartUpPosition = 0
    If side = "left" Then
        l = Application.Left + 12
    ElseIf side = "center" Then
        l = Application.Left + (Application.Width - frm.Width) / 2
    Else
        l = Application.Left + Application.Width - frm.Width - 16
    End If
    If l < Application.Left Then l = Application.Left
    frm.Left = l
    If side = "center" Then
        frm.Top = Application.Top + 120
    Else
        frm.Top = Application.Top + 4
    End If
End Sub

Public Sub UiFixHeight(ByVal frm As Object, ByVal contentBottom As Single)
    Dim want As Single
    Dim avail As Single
    want = contentBottom + (frm.Height - frm.InsideHeight)
    avail = Application.Height - 60
    If avail < 300 Then avail = 300
    If want > avail Then
        frm.Height = avail
        frm.ScrollBars = 2
        frm.KeepScrollBarsVisible = 0
        frm.ScrollHeight = contentBottom
    Else
        frm.Height = want
    End If
End Sub

                  ByVal nb As Long, ByVal nbd As Long, ByVal hb As Long, ByVal hbd As Long, _

    ev.NormalBack = nb
    ev.NormalBorder = nbd
    ev.HoverBack = hb
    ev.HoverBorder = hbd
    ev.Passive = passive
    keep.Add ev
End Sub

Private Function SettingsPath() As String
    SettingsPath = Environ$("APPDATA") & "\SPED\settings.txt"
End Function

Public Function Sped_GetSetting(ByVal k As String, ByVal def As String) As String
    Dim f As Integer
    Dim s As String
    Dim p As Long
    Sped_GetSetting = def
    If Len(Environ$("APPDATA")) = 0 Then Exit Function
    On Error GoTo done
    If Len(Dir(SettingsPath())) = 0 Then Exit Function
    f = FreeFile
    Open SettingsPath() For Input As #f
    Do While Not EOF(f)
        Line Input #f, s
        p = InStr(s, "=")
        If p > 1 Then
            If LCase$(Trim$(Left$(s, p - 1))) = LCase$(k) Then Sped_GetSetting = Mid$(s, p + 1)
        End If
    Loop
done:
    On Error Resume Next
    Close #f
End Function

Public Sub Sped_SetSetting(ByVal k As String, ByVal v As String)
    Dim f As Integer
    Dim folder As String
    Dim s As String
    Dim p As Long
    Dim keys() As String
    Dim vals() As String
    Dim n As Long
    Dim i As Long
    Dim found As Boolean

    On Error GoTo fail
    folder = Environ$("APPDATA") & "\SPED"
    If Len(Dir(folder, vbDirectory)) = 0 Then MkDir folder
    ReDim keys(1 To 64)
    ReDim vals(1 To 64)
    If Len(Dir(SettingsPath())) > 0 Then
        f = FreeFile
        Open SettingsPath() For Input As #f
        Do While Not EOF(f)
            Line Input #f, s
            p = InStr(s, "=")
            If p > 1 And n < 63 Then
                n = n + 1
                keys(n) = Trim$(Left$(s, p - 1))
                vals(n) = Mid$(s, p + 1)
            End If
        Loop
        Close #f
    End If
    For i = 1 To n
        If LCase$(keys(i)) = LCase$(k) Then
            vals(i) = v
            found = True
        End If
    Next i
    If Not found Then
        n = n + 1
        keys(n) = k
        vals(n) = v
    End If
    f = FreeFile
    Open SettingsPath() For Output As #f
    For i = 1 To n
        Print #f, keys(i) & "=" & vals(i)
    Next i
    Close #f
    Exit Sub
fail:
    On Error Resume Next
    Close #f
    MsgBox "Could not save the setting: " & Err.Description, vbExclamation, "SPED"
End Sub

Private Sub LoadTheme()

End Sub

Public Function Fav_Has(ByVal key As String) As Boolean
    Fav_Has = (InStr("," & Fav_List() & ",", "," & key & ",") > 0)
End Function

Public Function Fav_Toggle(ByVal key As String) As Boolean
    ' Returns True when the key is pinned afterwards. Keeps the standard order.
    Dim allK As Variant
    Dim i As Long
    Dim cur As String
    Dim out As String
    Dim want As Boolean
    want = Not Fav_Has(key)
    cur = "," & Fav_List() & ","
    allK = Fav_AllKeys()
    For i = LBound(allK) To UBound(allK)
        If CStr(allK(i)) = key Then
            If want Then out = out & "," & key
        ElseIf InStr(cur, "," & CStr(allK(i)) & ",") > 0 Then
            out = out & "," & CStr(allK(i))
        End If
    Next i
    If Len(out) > 0 Then out = Mid$(out, 2)
    Sped_SetSetting "Favs", out
    Fav_Toggle = want
End Function
