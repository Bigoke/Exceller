Attribute VB_Name = "modAuto"
Option Explicit

'==========================================================================
' SPED - Auto: one click to tidy a sheet (clean text, fix text numbers and
' dates, one font, sensible alignment, auto-fit columns and rows).
'==========================================================================

Private Const AUTO_FONT As String = "Calibri"
Private Const AUTO_SIZE As Double = 10
Private Const MIN_W As Double = 6
Private Const MAX_W As Double = 50
Private Const MAX_CELLS As Double = 500000
Private Const MAX_CHANGES As Long = 100000

Private Function Tidy(ByVal s As String) As String
    Dim i As Long
    Dim j As Long
    s = Replace(s, ChrW(160), " ")
    s = Replace(s, ChrW(8203), "")
    s = Replace(s, ChrW(8204), "")
    s = Replace(s, ChrW(8205), "")
    s = Replace(s, ChrW(65279), "")
    s = Replace(s, ChrW(173), "")
    s = Replace(s, ChrW(65533), "")
    s = Replace(s, vbTab, " ")
    s = Replace(s, vbCr, "")
    For i = 0 To 31
        If i <> 10 Then s = Replace(s, Chr$(i), "")
    Next i
    s = Replace(s, Chr$(127), "")
    i = 1
    j = Len(s)
    Do While i <= j
        If Mid$(s, i, 1) = " " Or Mid$(s, i, 1) = vbLf Then i = i + 1 Else Exit Do
    Loop
    Do While j >= i
        If Mid$(s, j, 1) = " " Or Mid$(s, j, 1) = vbLf Then j = j - 1 Else Exit Do
    Loop
    Tidy = Mid$(s, i, j - i + 1)
End Function

Private Function AreaArr(ByVal ar As Range) As Variant
    Dim a() As Variant
    If ar.CountLarge = 1 Then
        ReDim a(1 To 1, 1 To 1)
        a(1, 1) = ar.Value
        AreaArr = a
    Else
        AreaArr = ar.Value
    End If
End Function

Private Function Show1(ByVal s As String) As String
    s = Replace(s, vbLf, "/")
    If Len(s) > 22 Then s = Left$(s, 21) & "~"
    Show1 = "[" & s & "]"
End Function

Private Function Examples(ByRef cs As SpedChange, ByVal maxN As Long) As String
    Dim i As Long
    Dim k As Long
    Dim o As String
    Dim n As String
    For i = 1 To cs.n
        If k >= maxN Then Exit For
        If VarType(cs.oldVal(i)) = vbString Then
            o = Show1(CStr(cs.oldVal(i)))
            If VarType(cs.newVal(i)) = vbString Then
                n = Show1(CStr(cs.newVal(i)))
            ElseIf InStr(cs.newNF(i), "yy") > 0 Then
                n = Format(CDate(cs.newVal(i)), "dd-mmm-yyyy")
            Else
                n = CStr(cs.newVal(i))
            End If
            Examples = Examples & cs.addr(i) & ": " & o & " -> " & n & vbCrLf
            k = k + 1
        End If
    Next i
End Function

Private Function IsFilled(ByVal x As Variant) As Boolean
    Dim t As String
    If IsEmpty(x) Then Exit Function
    If VarType(x) = vbString Then
        If Len(x) = 0 Then Exit Function
        If Left$(x, 1) = "=" Then
            IsFilled = True
            Exit Function
        End If
        t = Trim$(Replace(CStr(x), ChrW(160), " "))
        If Len(t) = 0 Then Exit Function
        If Len(t) <= 3 Then
            IsFilled = (Len(Tidy(t)) > 0)
        Else
            IsFilled = True
        End If
    Else
        IsFilled = True
    End If
End Function

Private Function FindHeader(ByVal ws As Worksheet, ByVal rng As Range) As Long
    Dim r As Long
    Dim lastRow As Long
    Dim cnt As Long
    Dim firstNon As Long
    Dim slice As Range
    lastRow = rng.Row + rng.Rows.Count - 1
    For r = rng.Row To lastRow
        If r > rng.Row + 14 Then Exit For
        Set slice = Intersect(rng, ws.Rows(r))
        cnt = Application.WorksheetFunction.CountA(slice)
        If cnt >= 2 Then
            FindHeader = r
            Exit Function
        ElseIf cnt = 1 And firstNon = 0 Then
            firstNon = r
        End If
    Next r
    FindHeader = firstNon
End Function

' Rows / columns that are completely empty on the whole sheet (inside the used range and the scope).
Private Sub FindEmpties(ByVal ws As Worksheet, ByVal rng As Range, ByRef delRows() As Long, ByRef nRows As Long, ByRef delCols() As Long, ByRef nCols As Long)
    Dim ur As Range
    Dim ar() As Variant
    Dim i As Long
    Dim j As Long
    Dim nr As Long
    Dim nc As Long
    Dim rowUsed() As Boolean
    Dim colUsed() As Boolean
    Dim r As Long
    Dim c As Long
    Dim m As Variant
    Dim h As Variant

    nRows = 0
    nCols = 0
    Set ur = ws.UsedRange
    nr = ur.Rows.Count
    nc = ur.Columns.Count
    ReDim rowUsed(1 To nr)
    ReDim colUsed(1 To nc)
    ReDim delRows(1 To nr)
    ReDim delCols(1 To nc)
    If ur.CountLarge = 1 Then
        ReDim ar(1 To 1, 1 To 1)
        ar(1, 1) = ur.Formula
    Else
        ar = ur.Formula
    End If
    For i = 1 To nr
        For j = 1 To nc
            If IsFilled(ar(i, j)) Then
                rowUsed(i) = True
                colUsed(j) = True
            End If
        Next j
    Next i
    For i = 1 To nr
        If Not rowUsed(i) Then
            r = ur.Row + i - 1
            If r >= rng.Row And r <= rng.Row + rng.Rows.Count - 1 Then
                h = ws.Rows(r).Hidden
                m = ws.Rows(r).MergeCells
                If h = False And Not IsNull(m) Then
                    If m = False Then
                        nRows = nRows + 1
                        delRows(nRows) = r
                    End If
                End If
            End If
        End If
    Next i
    For j = 1 To nc
        If Not colUsed(j) Then
            c = ur.Column + j - 1
            If c >= rng.Column And c <= rng.Column + rng.Columns.Count - 1 Then
                h = ws.Columns(c).Hidden
                m = ws.Columns(c).MergeCells
                If h = False And Not IsNull(m) Then
                    If m = False Then
                        nCols = nCols + 1
                        delCols(nCols) = c
                    End If
                End If
            End If
        End If
    Next j
End Sub

Private Sub DeleteBlocks(ByVal ws As Worksheet, ByRef a() As Long, ByVal n As Long, ByVal isRow As Boolean)
    Dim i As Long
    Dim e As Long
    Dim st As Long
    i = n
    Do While i >= 1
        e = a(i)
        st = e
        Do While i > 1
            If a(i - 1) = st - 1 Then
                st = st - 1
                i = i - 1
            Else
                Exit Do
            End If
        Loop
        If isRow Then
            ws.Rows(st & ":" & e).Delete
        Else
            ws.Range(ws.Columns(st), ws.Columns(e)).Delete
        End If
        i = i - 1
    Loop
End Sub

Public Sub Auto_Run()
    Dim ws As Worksheet
    Dim rng As Range
    Dim scopeTxt As String
    Dim tc As Range
    Dim tn As Range
    Dim ar As Range
    Dim arr As Variant
    Dim i As Long
    Dim j As Long
    Dim s0 As String
    Dim t As String
    Dim v As Double
    Dim pct As Boolean
    Dim skz As Boolean
    Dim dt As Date
    Dim amb As Boolean
    Dim handled As Boolean
    Dim cs As SpedChange
    Dim c As Range
    Dim nf As Variant
    Dim nTrim As Long
    Dim nNum As Long
    Dim nDate As Long
    Dim nAmb As Long
    Dim nFmt As Long
    Dim nfNew As String
    Dim hdrRow As Long
    Dim lastRow As Long
    Dim r As Long
    Dim slice As Range
    Dim cnt As Long
    Dim fitRng As Range
    Dim vis As Range
    Dim a As Range
    Dim calc As Long
    Dim msg As String
    Dim hdr As Range
    Dim w As Double
    Dim delRows() As Long
    Dim delCols() As Long
    Dim nDelR As Long
    Dim nDelC As Long
    Dim refs() As Range
    Dim cs2 As SpedChange
    Dim a2 As String
    Dim keep As Boolean
    Dim kept As Long

    On Error GoTo bail
    If TypeName(ActiveSheet) <> "Worksheet" Then
        MsgBox "Activate a worksheet first.", vbExclamation, "SPED - Auto"
        Exit Sub
    End If
    Set ws = ActiveSheet
    If ws.ProtectContents Then
        MsgBox "This sheet is protected. Unprotect it first.", vbExclamation, "SPED - Auto"
        Exit Sub
    End If
    If Application.WorksheetFunction.CountA(ws.UsedRange) = 0 Then
        MsgBox "The sheet is empty.", vbInformation, "SPED - Auto"
        Exit Sub
    End If

    If TypeName(Selection) = "Range" Then
        If Selection.CountLarge > 1 Then
            If Selection.Areas.Count > 1 Then
                MsgBox "Please select one block of cells, or a single cell to do the whole sheet.", vbExclamation, "SPED - Auto"
                Exit Sub
            End If
            Set rng = Intersect(Selection, ws.UsedRange)
            If rng Is Nothing Then
                MsgBox "There is no data in the selection.", vbInformation, "SPED - Auto"
                Exit Sub
            End If
            scopeTxt = "selection"
        End If
    End If
    If rng Is Nothing Then
        Set rng = ws.UsedRange
        scopeTxt = "whole sheet"
    End If
    If rng.CountLarge > MAX_CELLS Then
        MsgBox "That is more than " & Format(MAX_CELLS, "#,##0") & " cells. Select a smaller block.", vbExclamation, "SPED - Auto"
        Exit Sub
    End If

    Application.StatusBar = "SPED Auto: checking the data..."
    Application.ScreenUpdating = False

    '---- 1. text: clean, then recognise numbers and dates ---------------
    Set tc = Nothing
    On Error Resume Next
    Set tc = rng.SpecialCells(xlCellTypeConstants, xlTextValues)
    On Error GoTo bail
    If Not tc Is Nothing Then
        For Each ar In tc.Areas
            arr = AreaArr(ar)
            For i = 1 To UBound(arr, 1)
                For j = 1 To UBound(arr, 2)
                    If VarType(arr(i, j)) = vbString Then
                        If cs.n >= MAX_CHANGES Then GoTo doneText
                        s0 = CStr(arr(i, j))
                        t = Tidy(s0)
                        handled = False
                        If Len(t) > 0 Then
                            If ParseNumberText(t, 3, v, pct, skz) Then
                                If pct Then
                                    nfNew = "0.00%"
                                ElseIf InStr(t, ".") > 0 Or InStr(t, ",") > 0 Or InStr(t, "(") > 0 Then
                                    nfNew = "#,##0.00"
                                Else
                                    nfNew = "General"
                                End If
                                Set c = ar.Cells(i, j)
                                Chg_Add cs, c.Address(False, False), s0, v, CStr(c.NumberFormat), nfNew
                                nNum = nNum + 1
                                handled = True
                            ElseIf ParseDateText(t, True, dt, amb) Then
                                If dt <> Int(dt) Then nfNew = "dd-mmm-yyyy hh:nn" Else nfNew = "dd-mmm-yyyy"
                                Set c = ar.Cells(i, j)
                                Chg_Add cs, c.Address(False, False), s0, CDbl(dt), CStr(c.NumberFormat), nfNew
                                nDate = nDate + 1
                                If amb Then nAmb = nAmb + 1
                                handled = True
                            End If
                        End If
                        If Not handled And t <> s0 Then
                            Set c = ar.Cells(i, j)
                            Chg_Add cs, c.Address(False, False), s0, t, CStr(c.NumberFormat), ""
                            nTrim = nTrim + 1
                        End If
                    End If
                Next j
            Next i
        Next ar
    End If
doneText:

    '---- 2. real numbers stuck in Text format ----------------------------
    Set tn = Nothing
    On Error Resume Next
    Set tn = rng.SpecialCells(xlCellTypeConstants, xlNumbers)
    On Error GoTo bail
    If Not tn Is Nothing Then
        For Each ar In tn.Areas
            If cs.n >= MAX_CHANGES Then Exit For
            nf = ar.NumberFormat
            If IsNull(nf) Then
                For Each c In ar.Cells
                    If c.NumberFormat = "@" And cs.n < MAX_CHANGES Then
                        Chg_Add cs, c.Address(False, False), c.Value2, c.Value2, "@", "General"
                        nFmt = nFmt + 1
                    End If
                Next c
            ElseIf CStr(nf) = "@" Then
                For Each c In ar.Cells
                    If cs.n < MAX_CHANGES Then
                        Chg_Add cs, c.Address(False, False), c.Value2, c.Value2, "@", "General"
                        nFmt = nFmt + 1
                    End If
                Next c
            End If
        Next ar
    End If

    lastRow = rng.Row + rng.Rows.Count - 1
    hdrRow = FindHeader(ws, rng)
    FindEmpties ws, rng, delRows, nDelR, delCols, nDelC

    '---- confirm -----------------------------------------------------------
    msg = "Auto format on '" & ws.Name & "' (" & scopeTxt & ", " & rng.Address(False, False) & "):" & vbCrLf & vbCrLf
    msg = msg & "- Clean spaces / hidden characters: " & nTrim & " cell(s)" & vbCrLf
    msg = msg & "- Text to numbers: " & nNum & "   Text to dates: " & nDate & IIf(nAmb > 0, " (" & nAmb & " unclear, read day/month/year)", "") & vbCrLf
    msg = msg & "- Numbers stuck in Text format fixed: " & nFmt & vbCrLf
    msg = msg & "- Font " & AUTO_FONT & " " & AUTO_SIZE & ", text left, numbers right" & IIf(hdrRow > 0, ", heading row " & hdrRow & " centred and bold", "") & vbCrLf
    msg = msg & "- Delete empty rows: " & nDelR & "   Delete empty columns: " & nDelC & vbCrLf
    If nDelR + nDelC > 0 Then msg = msg & "  (formulas that point straight at a deleted empty cell will show #REF!)" & vbCrLf
    msg = msg & "- Fit column widths (" & MIN_W & " to " & MAX_W & ") and row heights" & vbCrLf
    If cs.n > 0 Then msg = msg & vbCrLf & "Examples:" & vbCrLf & Examples(cs, 6)
    msg = msg & vbCrLf & "Value changes can be undone with SPED > Undo. Deleted rows and columns, fonts, alignment and widths cannot." & vbCrLf & "Apply now?"
    Application.ScreenUpdating = True
    Application.StatusBar = False
    If MsgBox(msg, vbYesNo + vbQuestion, "SPED - Auto") = vbNo Then Exit Sub

    '---- apply ---------------------------------------------------------------
    calc = Application.Calculation
    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    Application.StatusBar = "SPED Auto: working..."

    If cs.n > 0 Then ReDim refs(1 To cs.n)
    For i = 1 To cs.n
        Set c = ws.Range(cs.addr(i))
        Set refs(i) = c
        If VarType(cs.newVal(i)) = vbString Then
            WriteText c, CStr(cs.newVal(i))
        Else
            If Len(cs.newNF(i)) > 0 Then c.NumberFormat = cs.newNF(i)
            c.Value2 = cs.newVal(i)
        End If
    Next i

    ' delete empty rows / columns, then re-map the undo addresses
    If nDelR > 0 Or nDelC > 0 Then
        If nDelR > 0 Then DeleteBlocks ws, delRows, nDelR, True
        If nDelC > 0 Then DeleteBlocks ws, delCols, nDelC, False
        For i = 1 To cs.n
            keep = False
            On Error Resume Next
            a2 = refs(i).Address(False, False)
            If Err.Number = 0 Then
                If Len(a2) > 0 And InStr(a2, "REF") = 0 Then keep = True
            End If
            Err.Clear
            On Error GoTo bail
            If keep Then
                Chg_Add cs2, a2, cs.oldVal(i), cs.newVal(i), cs.oldNF(i), cs.newNF(i)
                kept = kept + 1
            End If
        Next i
        cs = cs2
        If kept = 0 Then cs.n = 0
        On Error Resume Next
        a2 = rng.Address
        If Err.Number <> 0 Or scopeTxt = "whole sheet" Then
            Err.Clear
            Set rng = ws.UsedRange
        End If
        Err.Clear
        On Error GoTo bail
        lastRow = rng.Row + rng.Rows.Count - 1
        hdrRow = FindHeader(ws, rng)
    End If
    If cs.n > 0 Then Undo_PushCells "Auto format (" & cs.n & " cells on " & ws.Name & ")", ws, cs

    rng.Font.Name = AUTO_FONT
    rng.Font.Size = AUTO_SIZE
    rng.VerticalAlignment = xlCenter
    rng.HorizontalAlignment = xlLeft
    On Error Resume Next
    rng.SpecialCells(xlCellTypeConstants, xlNumbers).HorizontalAlignment = xlRight
    rng.SpecialCells(xlCellTypeFormulas, xlNumbers).HorizontalAlignment = xlRight
    On Error GoTo bail
    If hdrRow > 0 Then
        Set hdr = Intersect(rng, ws.Rows(hdrRow))
        hdr.HorizontalAlignment = xlCenter
        hdr.Font.Bold = True
    End If

    ' column widths: ignore title lines above the heading row
    Set fitRng = rng
    If hdrRow > rng.Row Then
        Set fitRng = ws.Range(ws.Cells(hdrRow, rng.Column), ws.Cells(lastRow, rng.Column + rng.Columns.Count - 1))
    End If
    Set vis = Nothing
    On Error Resume Next
    Set vis = fitRng.SpecialCells(xlCellTypeVisible)
    On Error GoTo bail
    If Not vis Is Nothing Then
        For Each a In vis.Areas
            a.Columns.AutoFit
        Next a
    End If
    For j = 1 To rng.Columns.Count
        If Not rng.Columns(j).EntireColumn.Hidden Then
            w = rng.Columns(j).ColumnWidth
            If w > MAX_W Then
                rng.Columns(j).ColumnWidth = MAX_W
            ElseIf w < MIN_W Then
                rng.Columns(j).ColumnWidth = MIN_W
            End If
        End If
    Next j

    Set vis = Nothing
    On Error Resume Next
    Set vis = rng.SpecialCells(xlCellTypeVisible)
    On Error GoTo bail
    If Not vis Is Nothing Then
        For Each a In vis.Areas
            a.Rows.AutoFit
        Next a
    End If

    Application.Calculation = calc
    Application.ScreenUpdating = True
    Application.StatusBar = "SPED Auto: " & ws.Name & " tidied (" & cs.n & " value change(s))."
    Log_Add "Auto format", ws.Name & " " & rng.Address(False, False) & ": " & nTrim & " cleaned, " & nNum & " numbers, " & nDate & " dates, " & nFmt & " formats, " & nDelR & " empty rows and " & nDelC & " empty columns deleted"
    Exit Sub
bail:
    On Error Resume Next
    Application.Calculation = calc
    Application.ScreenUpdating = True
    Application.StatusBar = False
    MsgBox "Auto stopped: " & Err.Description, vbExclamation, "SPED - Auto"
End Sub
