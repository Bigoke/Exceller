Private Sub BuildFavs()
    Dim keys() As String
    Dim i As Long
    Dim n As Long
    Dim x As Single
    Dim g As Long
    Dim cp As String
    Dim ac As String
    keys = Split(Fav_List(), ",")
    For i = 0 To UBound(keys)
        If Fav_Info(Trim$(keys(i)), g, cp, ac) Then n = n + 1
    Next i
    Me.Caption = "SPED"
    UiStyleForm Me, (n + 1) * 32 + 28, 80
    x = 8
    For i = 0 To UBound(keys)
        If Fav_Info(Trim$(keys(i)), g, cp, ac) Then
            UiIconButton Me, "fv" & Trim$(keys(i)), g, Replace(cp, "&&", "&"), ac, x, 6, 28, mKeep

        End If
    Next i
    UiIconButton Me, "fvedit", &H2699, "Edit favourites", "Act_FavEditFromBar", x, 6, 28, mKeep
    UiFixHeight Me, 40

Private Sub FillFormats()
    Dim names As Variant
    Dim i As Long
    Dim cur As String
    If cboFmt Is Nothing Then Exit Sub
    mFilling = True
    cboFmt.Clear
    cboFmt.AddItem "(not applying)"
    names = Cf_FormatNames()
    If Not IsEmpty(names) Then
        For i = LBound(names) To UBound(names)
            cboFmt.AddItem CStr(names(i))
        Next i
    End If
    cboFmt.ListIndex = 0
    cur = Cf_ActiveName()
    If Len(cur) > 0 Then
        For i = 1 To cboFmt.ListCount - 1
            If StrComp(cboFmt.List(i), cur, vbTextCompare) = 0 Then cboFmt.ListIndex = i
        Next i
    End If
    mFilling = False
End Sub

Private Sub cboFmt_Change()
    If mFilling Then Exit Sub
    If cboFmt.ListIndex <= 0 Then
        Cf_Stop
    Else
        Cf_PickByName cboFmt.List(cboFmt.ListIndex)
    End If
End Sub

Public Sub Act_DeleteRowsByValue()


    Dim rng As Range
    Dim cell As Range
    Dim searchValue As String
    Dim delRange As Range
    
    ' Check that a range is selected
    If TypeName(Selection) <> "Range" Then
        MsgBox "Please select a range first.", vbExclamation
        Exit Sub
    End If
    
    ' Ask the user what value to look for
    searchValue = InputBox("Enter the value to delete rows for." & vbCrLf & _
                           "Leave blank to delete rows with empty cells:", _
                           "Delete Rows by Value")
    
    ' If user clicks Cancel, exit
    If searchValue = "" Then
        If MsgBox("You entered nothing. This will delete rows with empty cells. Continue?", vbYesNo + vbExclamation) = vbNo Then Exit Sub
    End If
    
    ' Loop through the selected range
    Set rng = Intersect(Selection, ActiveSheet.UsedRange)
    If rng Is Nothing Then
        MsgBox "No matching cells found in selection.", vbInformation
        Exit Sub
    End If
    For Each cell In rng
        If IsError(cell.Value) Then
            ' skip error cells
        ElseIf (searchValue <> "" And CStr(cell.Value) = searchValue) Or (searchValue = "" And IsEmpty(cell.Value)) Then
            If delRange Is Nothing Then
                Set delRange = cell
            Else
                Set delRange = Union(delRange, cell)
            End If
        End If
    Next cell
    
    ' Delete all matching rows
    If Not delRange Is Nothing Then
        delRange.EntireRow.Delete
        If searchValue = "" Then
            MsgBox "Rows with empty cells have been deleted.", vbInformation
        Else
            MsgBox "Rows with value '" & searchValue & "' have been deleted.", vbInformation
        End If
    Else
        MsgBox "No matching cells found in selection.", vbInformation
    End If



End Sub

Public Sub Act_CopySheet()
Dim ws As Worksheet
    Dim newName As String
    Dim nameExists As Boolean
    Dim sh As Worksheet
    
    Set ws = ActiveSheet
    
    ' Ask user for sheet name
    newName = InputBox("Enter the name for the new sheet:", "New Sheet Name")
    
    ' If user cancels or leaves blank
    If Trim(newName) = "" Then
        MsgBox "Operation cancelled.", vbExclamation
        Exit Sub
    End If
    
    ' Check if name already exists
    nameExists = False
    For Each sh In ThisWorkbook.Worksheets
        If LCase(sh.Name) = LCase(newName) Then
            nameExists = True
            Exit For
        End If
    Next sh
    
    If nameExists Then
        MsgBox "A sheet with this name already exists.", vbCritical
        Exit Sub
    End If
    
    ' Copy sheet after current sheet
    ws.Copy After:=ws
    
    ' Rename the copied sheet
    ActiveSheet.Name = newName
    
    MsgBox "Sheet created successfully!", vbInformation

End Sub

Public Sub Act_NewRow()
' NewRow Macro
'

'
    If MsgBox("Do you want to add a new row?", vbYesNo) = vbNo Then Exit Sub
    Selection.EntireRow.Insert , CopyOrigin:=xlFormatFromLeftOrAbove
    If MsgBox("Do you want to link a account as well?", vbYesNo) = vbNo Then Exit Sub

    Selection.Copy
    ActiveCell.Worksheet.Cells(ActiveCell.Row + 1, 1).EntireRow.Select
    ActiveSheet.Paste
    Application.CutCopyMode = False
End Sub

End Sub
